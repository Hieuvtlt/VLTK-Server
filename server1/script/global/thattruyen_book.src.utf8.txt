-- Bi kip Vo hoc That Truyen - script VAT PHAM
-- Dat tai: \script\global\thattruyen_book.lua
--
-- Vat pham {6, 1, TT_BOOK_P} - khai bao trong settings/item/magicscript.txt,
-- cot script tro toi chinh file nay.
--
-- Su phu chi TRAO bi kip (thattruyen_head.lua -> tt_do_learn).
-- Doc bi kip moi thuc su duoc skill - va van phai kiem tra lai tuyet ky bon mon,
-- vi nguoi choi co the cat bi kip roi doc rat lau ve sau.
--
-- Quy uoc tra ve cua script vat pham (theo mau lv120skillbook.lua):
--   return 0 -> dung thanh cong, engine TIEU vat pham
--   return 1 -> khong dung duoc, GIU nguyen vat pham

Include("\\script\\global\\thattruyen_head.lua")

function main(idxItem)

	local nFaction = GetLastFactionNumber()

	if (nFaction < 0) or (nFaction > 9) then
		Msg2Player("<color=yellow>Chưa gia nhập môn phái thì không đọc nổi bí kíp này.<color>")
		return 1
	end

	-- KHOA: phai co "Bac Minh tam phap" trong hanh trang moi luyen duoc.
	-- Kiem TRUOC khi trao bat cu skill nao, va GIU nguyen bi kip (return 1).
	if (CalcItemCount(3, 6, 1, TT_KEY_P, -1) < 1) then
		Say("Cuốn bí kíp chỉ có <color=yellow>khẩu quyết<color>, nhưng hoàn toàn không có <color=yellow>tâm pháp<color> vận khí."
			.."<enter>Biết cách ra chiêu nhưng không biết vận công."
			.."<enter>Biết cách dẫn khí nhưng không biết khí phải đi đâu."
			.."<enter>Biết võ chiêu nhưng không thể sinh ra nội lực tương ứng."
			.."<enter>Càng cố luyện, kinh mạch càng đau đớn.",
			1, "Kết thúc/tt_book_quit")
		return 1
	end

	local tList = TT_SKILL_LIST[nFaction]
	local nLearned, nAlready, nLack = 0, 0, 0
	local szLearned = ""

	for i = 1, getn(tList) do
		local nTT     = tList[i][1]
		local nOrigin = tList[i][2]

		if (HaveMagic(nTT) ~= -1) then
			nAlready = nAlready + 1
		elseif (HaveMagic(nOrigin) >= TT_ORIGIN_MAXLEVEL) then
			AddMagic(nTT, TT_SKILL_LEVEL)
			nLearned = nLearned + 1
			szLearned = szLearned.." "..nTT
		else
			nLack = nLack + 1
		end
	end

	-- Khong hoc duoc gi -> GIU lai bi kip, giai thich ro ly do
	if (nLearned == 0) then
		if (nAlready >= getn(tList)) then
			Msg2Player("<color=yellow>Ngươi đã lĩnh hội trọn vẹn bộ võ học thất truyền rồi.<color>")
		else
			Msg2Player("<color=yellow>Tuyệt kỹ bổn môn chưa luyện đến cấp "
				..TT_ORIGIN_MAXLEVEL..", chưa thể lĩnh hội. Bí kíp vẫn còn nguyên.<color>")
		end
		return 1
	end

	WriteLog(GetLocalDate("%Y-%m-%d %X").."\t[ThatTruyen-DocBiKip]\tAccount:"..GetAccount()
		.."\tName:"..GetName().."\tFaction:"..nFaction
		.."\tHoc:"..nLearned.."\tSkillId:"..szLearned
		.."\tDaCo:"..nAlready.."\tConThieuGoc:"..nLack)

	Msg2Player("<color=yellow>Ngươi đã lĩnh hội "..nLearned
		.." môn võ học thất truyền của bổn môn!<color>")

	-- Con mon chua mo khoa thi bao de nguoi choi khong tuong bi kip bi mat oan
	if (nLack > 0) then
		Msg2Player("<color=red>Còn "..nLack
			.." môn chưa lĩnh hội được vì tuyệt kỹ bổn môn chưa luyện thành."
			.." Hãy xin sư phụ bí kíp mới sau khi luyện xong.<color>")
	end

	Say("Hai cuốn bí kíp đặt cạnh nhau. Điều kỳ lạ xảy ra, những câu chữ tưởng như hoàn toàn không liên quan bắt đầu bổ sung cho nhau."
		.."<enter>Khẩu quyết của cuốn thứ nhất... Lại chính là những \"điểm khuyết\" mà tâm pháp của cuốn thứ hai có thể lấp đầy."
		.."<enter>Một cuốn là thân ... Một cuốn là tâm !"
		.."<enter>Chúng vốn được sinh ra để kết hợp với nhau."
		.."<enter>Ngươi không hề học một võ công có sẵn."
		.."<enter><color=yellow>Ngươi đang tạo ra một võ công mới từ hai thứ võ học vốn không hoàn chỉnh.<color>",
		1, "Kết thúc/tt_book_quit")

	return 0

end

function tt_book_quit()
end
